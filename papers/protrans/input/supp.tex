\paragraph{Quantitative constant convention.}
All constants in quantitative upper bounds and probability-decay exponents
are strictly positive unless explicitly assigned another range. They depend
only on the fixed regularity parameters specified in the relevant statement,
not silently on sample sizes or dimensions. This convention does not impose
signs on model coefficients or algebraically defined quantities.

\section{Detailed Proofs and Technique Lemmas}
\subsection{Proof of Theorem 1}
\paragraph{Notation and constant ranges.}
In this subsection, Assumption 1 means Assumption
\ref{assum:distribution_E_H_f} of the main paper, including its uniform
sub-Gaussian and bounded-heterogeneity conditions. All concentration
constants and confidence exponents are strictly positive. Fix the requested
exponent $c>0$ before choosing the constants. Put
\[
 w_k=n_s^{(k)}/n_s,\quad a_t=\sqrt{\log p/n_t},\quad
 a_s=\sqrt{\log p/n_s},\quad
 \Sigma_X=\mathbb E[XX^\top],\quad M=\mathbb E[XH^\top].
\]
For a matrix, $\|\cdot\|_\infty$ denotes the largest absolute entry; for a
vector it is the usual maximum norm. We use
$\|B\|_{2,\infty}=\max_j\|B_{j,:}\|_2$.
The moments exist by the uniform sub-Gaussian assumptions. In particular,
$M=\mathbb E[f(H)H^\top]$ because $\mathbb E[EH^\top]=0$, and there is a
fixed $L_M>0$, depending only on $L$, with $\|M\|_{2,\infty}\le L_M\sqrt q$.
This follows coordinatewise from Cauchy--Schwarz and the second-moment
bounds. We work with $n_s\ge n_t\ge N_0\log p$ and choose $N_0\ge1$ large
enough for the concentration lemmas below; consequently $a_s\le a_t\le1$.

\begin{pf}{\it of Theorem 1.}
For later cross-references, the concentration events are
    \begin{align}
       & \left\{\left\|\frac{1}{n_t} X_t^\top X_t - \Sigma_X\right\|_{\infty} \le C_x\sqrt{\frac{\log p}{n_t}}\right\},\label{set:g_1}\tag{$\mathcal{G}_1$}\\
        &\left\{\left\|\frac{1}{n_s}\sum_{k=1}^K X_s^{(k)\top}X_s^{(k)} - \Sigma_X\right\|_{\infty} \le C_x\sqrt{\frac{\log p}{n_s}}\right\},\label{set:g_2}\tag{$\mathcal{G}_2$} \\
        & \left\{ \left\|\frac{1}{n_s}\sum_{k=1}^K n_s^{(k)}\left(\frac{1}{n_s^{(k)}}X_s^{(k)\top}X_s^{(k)} - \Sigma_X \right)\left(\beta_s^{(k)} - \frac{1}{n_s}\sum_{k=1}^Kn_s^{(k)}\beta_s^{(k)}\right) \right\|_{\infty}\le  C_{\delta} s_\delta\sqrt{\frac{\log p}{n_s}} \right\},
          \label{set:g_3}\tag{$\mathcal{G}_3$}\\&
          \left\{\left\|\frac{1}{n_t}f(H_t)^\top H_t - M\right\|_{2,\infty} \le 
        C_{H} \sqrt{\frac{q \log p}{n_t}}\right\},\label{set:g_4}\tag{$\mathcal{G}_4$}\\&
        \left\{\left\|\frac{1}{n_s}\sum_{k=1}^K f(H_s^{(k)})^\top H_s^{(k)} - M \right\|_{2,\infty} \le
        C_{H} \sqrt{\frac{q \log p}{n_s}}\right\},\label{set:g_5}\tag{$\mathcal{G}_5$}\\& 
        \left\{\left\| \frac{1}{n_t} E_t^\top H_t\right\|_{2,\infty} \le C_H\sqrt{\frac{q\log p}{n_t}}\right\}, \label{set:g_6}\tag{$\mathcal{G}_6$}\\&
        \left\{\left\| \frac{1}{n_s} \sum_{k=1}^K E_s^{(k)\top} H_s^{(k)}\right\|_{2,\infty} \le C_H\sqrt{\frac{q\log p}{n_s}}\right\}, \label{set:g_7}\tag{$\mathcal{G}_7$}\\&
        \left\{\left\|\frac{1}{n_t}X_t^\top \varepsilon_t\right\|_{\infty} \le C_\varepsilon \sqrt{\frac{\log p}{n_t}}\right\}, \label{set:g_8}\tag{$\mathcal{G}_8$}\\&
        \left\{\left\|\frac{1}{n_s}\sum_{k=1}^K X_s^{(k)\top} \varepsilon_s^{(k)}\right\|_{\infty} \le C_\varepsilon \sqrt{\frac{\log p}{n_s}}\right\}, \label{set:g_9}\tag{$\mathcal{G}_9$}
    \end{align}
\begin{equation}
 \left\{\|T_\phi\|_\infty\le C_dB_\phi\sqrt q\,a_s\right\}.
 \label{set:g_10_weighted}\tag{$\mathcal G_{10}^{\phi}$}
\end{equation}
Here $T_\phi$ has the explicit definition below. The simultaneous event
$\mathcal G$ is the intersection of these ten events. The original labels
$\mathcal G_1,\ldots,\mathcal G_9$ are retained for later cross-references.

Let $\mathcal G$ be the simultaneous event described in Lemma
\ref{lem:high_probability_set}. On this event the target and pooled source
Gram matrices $G_t=X_t^\top X_t/n_t$ and
$G_s=\sum_kX_s^{(k)\top}X_s^{(k)}/n_s$ satisfy
\[
 \|G_t-\Sigma_X\|_\infty\le C_xa_t,\qquad
 \|G_s-\Sigma_X\|_\infty\le C_xa_s.
\]
Set $\delta_k=\beta_s^{(k)}-\beta_s$, $d_k=\phi_s^{(k)}-\phi_t$,
$\bar d=\sum_kw_kd_k$, and
\[
 T_\beta={1\over n_s}\sum_k
 (X_s^{(k)\top}X_s^{(k)}-n_s^{(k)}\Sigma_X)\delta_k,\qquad
 T_\phi={1\over n_s}\sum_k
 (X_s^{(k)\top}H_s^{(k)}-n_s^{(k)}M)d_k.
\]
The source coefficient definition gives $\sum_kw_k\delta_k=0$.
Lemmas \ref{lem:error_sample_covariance_sparse} and
\ref{lem:weighted_confounding_transfer} give
\[
 \|T_\beta\|_\infty\le C_\delta s_\delta a_s,\qquad
 \|T_\phi\|_\infty\le C_d B_\phi\sqrt q\,a_s.
\]
Also put $D_t=X_t^\top H_t/n_t-M$,
$D_s=\sum_kX_s^{(k)\top}H_s^{(k)}/n_s-M$,
$e_t=X_t^\top\varepsilon_t/n_t$ and
$e_s=\sum_kX_s^{(k)\top}\varepsilon_s^{(k)}/n_s$.
On $\mathcal G$, Lemmas \ref{lem:error_sample_covariance2} and
\ref{lem:error_noise} imply
\[
 \|D_t\|_{2,\infty}\le2C_H\sqrt q\,a_t,\quad
 \|D_s\|_{2,\infty}\le2C_H\sqrt q\,a_s,\quad
 \|e_t\|_\infty\le C_\varepsilon a_t,\quad
 \|e_s\|_\infty\le C_\varepsilon a_s.
\]
Crucially, the exact confounding decomposition is
\begin{align}
 A_t-A_s&=(D_t-D_s)\phi_t-M\bar d-T_\phi,
 \label{eq:correct_weighted_confounding_decomposition}\\
 A_t&=X_t^\top H_t\phi_t/n_t,\qquad
 A_s=\sum_kX_s^{(k)\top}H_s^{(k)}\phi_s^{(k)}/n_s.\notag
\end{align}
Thus average similarity controls the population term, while the independent
weighted-sum lemma controls its separate centered fluctuation:
\[
 \|A_t-A_s\|_\infty\le
 [4C_H\sqrt q\|\phi_t\|_2+C_dB_\phi\sqrt q+
 L_MC_\phi\sqrt q]a_t.
\]
One cannot replace the weighted centered sum by a pooled matrix multiplied
by an averaged coefficient vector.

Write $d=\beta_s-\hat\beta_{\mathrm{init}}$ and
$b=\sum_kX_s^{(k)\top}\hat Z_s^{(k)}/n_s$.
The source model yields the exact identity
\[
 b=G_sd+T_\beta+A_s+e_s.
\]
The four nonnegative error magnitudes may equivalently be named
\begin{align}
 B_1&=\|(G_t-G_s)d\|_\infty,\label{term:B_1}\tag{$B_1$}\\
 B_2&=\|T_\beta\|_\infty,\label{term:B_2}\tag{$B_2$}\\
 B_3&=\|A_t-A_s\|_\infty,\label{term:B_3}\tag{$B_3$}\\
 B_4&=\|e_s\|_\infty.\label{term:B_4}\tag{$B_4$}
\end{align}
These labels identify the corresponding covariance, source-spread,
confounding-transfer and source-noise estimates for subsequent proofs.
Use the feasible target profiling comparison point
$Z_0=X_td+H_t\phi_t$. Its target score is $G_td+A_t$, hence
\begin{align}
 \Delta &:={X_t^\top Z_0\over n_t}-b
       =(G_t-G_s)d-T_\beta+(A_t-A_s)-e_s,
 \label{eq:profiled_score_decomposition}\\
 \|\Delta\|_\infty&\le
 [4C_H\sqrt q\|\phi_t\|_2+B_0]a_t,\notag\\
 B_0&=2C_xC_s+C_\delta s_\delta+C_dB_\phi\sqrt q
       +L_MC_\phi\sqrt q+C_\varepsilon>0,\notag
\end{align}
where the inequality uses $\|d\|_1\le C_s$ on $\mathcal I$.
By the definition of the profiling minimizer,
$\|X_t^\top\hat Z_t/n_t-b\|_\infty\le\|\Delta\|_\infty$.
Therefore the full Lasso noise score obeys
\begin{align}
 Q&=\left\|{X_t^\top\over n_t}
       (X_td+H_t\phi_t-\hat Z_t+\varepsilon_t)\right\|_\infty
       \le2\|\Delta\|_\infty+\|e_t\|_\infty\notag\\
  &\le(C_1\sqrt q\|\phi_t\|_2+C_2)a_t,
 \label{eq:corrected_profiled_score_bound}
\end{align}
with the strictly positive choices $C_1=8C_H$ and
$C_2=2B_0+C_\varepsilon$. All dependencies are as stated in the theorem.
No asymptotic $o(a_t)$ term or unspecified relation $n_s\gg n_t$ is used.

For completeness the two minimizers exist in finite dimensions. The image
$\{X_t^\top Z/n_t:Z\in\mathbb R^{n_t}\}$ is a closed linear subspace;
minimizing its distance to $b$ in the maximum norm attains a minimum
by restricting to a bounded sublevel set in that image. Any preimage
supplies $\hat Z_t$. The Lasso objective is continuous and coercive because
$\lambda_t>0$, so it also attains a minimum.

Let $h=\hat\eta-\eta_0$. Lasso optimality and $Q\le\lambda_t/4$ give
\begin{align}
 {\|X_th\|_2^2\over n_t}
 &\le\lambda_t(\|\eta_0\|_1-\|\eta_0+h\|_1)
                  +{\lambda_t\over2}\|h\|_1\notag\\
 &\le{3\lambda_t\over2}\|h_{S_\eta}\|_1
                  -{\lambda_t\over2}\|h_{S_\eta^c}\|_1.
 \label{eq:sup:oracle_inequality}
\end{align}
Nonnegativity of the left side gives
$\|h_{S_\eta^c}\|_1\le3\|h_{S_\eta}\|_1$.
On $\mathcal R$, the RE bound therefore applies. If $h=0$ all conclusions
hold. Otherwise, by $\|h_{S_\eta}\|_1\le\sqrt{s_\eta}\|h\|_2$,
\[
 C_r\|h\|_2^2\le {3\lambda_t\over2}\sqrt{s_\eta}\|h\|_2,
 \qquad \|h\|_2\le{3\lambda_t\over2C_r}\sqrt{s_\eta}.
\]
The cone gives $\|h\|_1\le4\sqrt{s_\eta}\|h\|_2$, giving $6/C_r$.
Substituting the $\ell_2$ bound into
$\|X_th\|_2^2/n_t\le3\lambda_t\sqrt{s_\eta}\|h\|_2/2$
gives the prediction constant $3/(2\sqrt{C_r})$.
The case $s_\eta=0$ is included: the cone then forces $h=0$.
Finally, $\mathbb P(\mathcal G)\ge1-p^{-c}$; intersecting with
$\mathcal R$ and $\mathcal I$ gives exactly the probability statement in
the main theorem, without assuming these random events occur automatically.
\end{pf}

\subsection{Lemmas for Theorem 1}
\paragraph{Uniform concentration argument.}
We give the common scalar argument used in all five concentration bounds
below, including the weighted ones. There is a fixed $V>0$, depending only
on the uniform sub-Gaussian bounds, such that any centered product of two
coordinates among $E,H,f(H),X,\varepsilon$ has moments bounded by
$r!V^r$ for all integers $r\ge2$ (enlarge $V$ if necessary).
Indeed, the sub-Gaussian tail bound gives coordinate moments at most
$B\sqrt r$ in $L^r$; applying Cauchy--Schwarz to a product and then
centering gives a bound $(B' r)^r\le r!V^r$ after changing the constant.
For deterministic $u$ with $\|u\|_1\le D$, the same bound for a linear
combination of such centered products is $r!(VD)^r$, by the triangle
inequality in $L^r$. This argument requires no independence between the
two factors in a single product.

For $N$ independent centered variables $U_l$ with
$\mathbb E|U_l|^r\le r!V^r$, expanding the exponential series gives
$\log\mathbb E e^{\theta U_l}\le2V^2\theta^2$ for
$|\theta|V\le1/2$. Independence, exponential Markov, and optimization
with $\theta=\min(t/(4V^2),1/(2V))$ yield
\begin{equation}
 \mathbb P\left(\left|{1\over N}\sum_{l=1}^NU_l\right|>t\right)
 \le2\exp[-bN\min(t^2/V^2,t/V)]
 \label{eq:profiled_scalar_bernstein}
\end{equation}
for an absolute $b>0$. The same proof applies to independent variables
with different laws and the same moment bound. Consequently, for any
fixed confidence exponent $c>0$, constants $C,N_0>0$ can be chosen so
that deviations $C\sqrt{\log p/N}$ in at most $p^2$ or $pq$ coordinates
fail with probability at most $p^{-c}/20$, whenever
$N\ge N_0\log p$. Since $q$ is fixed, its factor in the union bound is
absorbed by the choice of $C,N_0$, not by changing $c$ silently.
For a normalization $D=0$ the corresponding random variable is identically
zero and its deviation bound is deterministic.

\begin{lemma}
\label{lem:error_sample_covariance1}
Under Assumption 1.a, for every $c>0$ there are $C_x,N_x>0$ such that for
$N\ge N_x\log p$ independent rows with the common covariate law,
\[
 \mathbb P\left(\|X^\top X/N-\Sigma_X\|_\infty
                 \le C_x\sqrt{\log p/N}\right)\ge1-p^{-c}/20.
\]
The same constants work for the target rows and for all pooled source rows.
\end{lemma}
\begin{pf}
Apply \eqref{eq:profiled_scalar_bernstein} to
$X_{li}X_{lj}-\mathbb E[X_iX_j]$ and take a union over $p^2$ entries.
The four terms in $X_iX_j=(f(H)_i+E_i)(f(H)_j+E_j)$ satisfy the uniform
moment bound, or use directly the sub-Gaussian bound for $X_i$.
In particular the covariance term for the $N\times p$ sample matrix
$f(H)$ is $f(H)^\top f(H)$, not $f(H)^\top H$; the latter
uses the $N\times q$ sample matrix $H$ and has different dimensions.
\end{pf}

\begin{lemma}
\label{lem:error_sample_covariance_sparse}
Under Assumption 1.a, let deterministic $\delta_k\in\mathbb R^p$ satisfy
$\|\delta_k\|_1\le s_\delta$, with $s_\delta\ge0$.
For every $c>0$ there are $C_\delta,N_\delta>0$ such that,
if $n_s\ge N_\delta\log p$,
\[
 \mathbb P\left(\left\|{1\over n_s}\sum_k
 (X_s^{(k)\top}X_s^{(k)}-n_s^{(k)}\Sigma_X)\delta_k
 \right\|_\infty\le C_\delta s_\delta a_s\right)\ge1-p^{-c}/20.
\]
No lower bound on each individual $n_s^{(k)}$ is needed.
\end{lemma}
\begin{pf}
For coordinate $j$, sum the independent centered scalar variables
$U_{ki,j}=X_{ki,j}(X_{ki}^\top\delta_k)-
\mathbb E[X_jX^\top]\delta_k$ over all $n_s$ source rows.
Their $L^r$ norms are bounded uniformly by a constant times
$s_\delta r$, using $\|\delta_k\|_1\le s_\delta$.
Normalize by $s_\delta$ when it is positive, apply
\eqref{eq:profiled_scalar_bernstein}, and take a union over $p$ coordinates.
If $s_\delta=0$ the assertion is deterministic.
This proof does not factor a sum of matrix--vector products into two sums.
\end{pf}

\begin{lemma}
\label{lem:error_sample_covariance2}
Under Assumption 1.a, put $M_f=\mathbb E[f(H)H^\top]$.
For every $c>0$ there are $C_H,N_H>0$ such that for
$N\ge N_H\log p$ independent common-law rows, simultaneously
\[
 \|f(H)^\top H/N-M_f\|_{2,\infty}
       \le C_H\sqrt{q\log p/N},\qquad
 \|E^\top H/N\|_{2,\infty}\le C_H\sqrt{q\log p/N}
\]
with probability at least $1-p^{-c}/20$.
\end{lemma}
\begin{pf}
The products $f(H)_jH_l$ and $E_jH_l$ obey the uniform scalar moment
bound. The latter are centered because $\mathbb E[EH^\top]=0$.
Take a union bound over the $2pq$ entries in the two matrices after applying
\eqref{eq:profiled_scalar_bernstein}. The maximum row norm is at most
$\sqrt q$ times the maximum absolute entry. This is an event inclusion;
the probability of a maximum is not the maximum of the probabilities.
\end{pf}

\begin{lemma}
\label{lem:weighted_confounding_transfer}
Under Assumption 1.a, let deterministic $d_k\in\mathbb R^q$ satisfy
$\|d_k\|_2\le B_\phi$ with $B_\phi>0$.
For every $c>0$ there are $C_d,N_d>0$ such that if
$n_s\ge N_d\log p$,
\[
 \mathbb P\left(\left\|{1\over n_s}\sum_k
 (X_s^{(k)\top}H_s^{(k)}-n_s^{(k)}M)d_k\right\|_\infty
       \le C_dB_\phi\sqrt q\,a_s\right)\ge1-p^{-c}/20.
\]
The constants depend only on the fixed distributional bounds and $c$;
$q$ is displayed explicitly in the threshold.
\end{lemma}
\begin{pf}
For each coordinate $j$ the summands are
$(X_{ki,j}H_{ki}^\top-\mathbb E[X_jH^\top])d_k$.
They are centered and independent across rows and source groups, even
though their deterministic vectors $d_k$ differ.
Since $\|d_k\|_1\le\sqrt q B_\phi$, the common moment argument applies
after division by $\sqrt q B_\phi$. Apply scalar Bernstein to all $n_s$
summands and take a union over $p$ coordinates. No $\log K$ or a minimum
individual source sample size enters the estimate.
\end{pf}

\begin{lemma}
\label{lem:error_noise}
Under Assumptions 1.a--b, for every $c>0$ there are
$C_\varepsilon,N_\varepsilon>0$ such that for
$N\ge N_\varepsilon\log p$ independent rows,
\[
 \mathbb P\left(\|X^\top\varepsilon/N\|_\infty
       \le C_\varepsilon\sqrt{\log p/N}\right)\ge1-p^{-c}/20.
\]
These constants work for both target and pooled source rows, including
zero noise variance.
\end{lemma}
\begin{pf}
Independence of $\varepsilon$ and $(E,H)$ and its zero mean imply
$\mathbb E[X_j\varepsilon]=0$. Products of sub-Gaussian variables satisfy
the common moment bound. Apply scalar Bernstein and a union over the $p$
coordinates. The constants use $L,L_\varepsilon$ and do not divide by
$\sigma$, so the case $\sigma=0$ causes no singularity.
\end{pf}

\begin{lemma}
\label{lem:high_probability_set}
Under Assumptions 1.a--b and the deterministic coefficient bounds in the
main theorem and Assumption 1.d, for every $c>0$ there exist strictly
positive $C_x,C_\delta,C_H,C_d,C_\varepsilon,N_0$ such that, whenever
$n_s\ge n_t\ge N_0\log p$, all the bounds used in the proof of Theorem 1
hold simultaneously on an event $\mathcal G$ with
$\mathbb P(\mathcal G)\ge1-p^{-c}$.
\end{lemma}
\begin{pf}
Use Lemma \ref{lem:error_sample_covariance1} twice, for target and pooled
source Gram matrices; Lemma \ref{lem:error_sample_covariance_sparse} once;
Lemma \ref{lem:error_sample_covariance2} twice; Lemma
\ref{lem:weighted_confounding_transfer} once; and Lemma
\ref{lem:error_noise} twice. There are eight failure events, each at most
$p^{-c}/20$. Choose $N_0\ge1$ to exceed the finitely many thresholds and
choose common positive constants at least as large as the corresponding
ones in those lemmas. The union bound gives failure at most
$8p^{-c}/20\le p^{-c}$. This step does not relabel a possibly negative
exponent as a positive constant. RE and initial-estimator events are kept
separate, as stated in the main theorem.
\end{pf}

\subsection{Proof of Proposition 1}
\begin{pf}{\it of Proposition 1. }
     We consider the special case that $\beta_s^{(k)} = \beta_s = 0,1\le k \le K$ and the noise distribution is Gaussian distribution $N(0,\sigma^2)$. We then consider the case that $f=0,\phi_t = 0$ and $X_t,X_s^{(k
     )}$ follow standard Gaussian distribution $N(0,I_p)$. Define the distribution of source data and target data as $P_s(\eta),P_t(\eta)$. For $\eta_1,\eta_2\in \mathcal{B}_0(s_\eta)$, We know that
    \begin{align*}
        D_{KL}\left(P_s(\eta_1)\|P_s(\eta_2)\right) = 0,\quad 
        D_{KL}\left(P_t(\eta_1)\|P_t(\eta_2)\right) = \frac{1}{2\sigma^2} \left\|X_t\left(\eta_1 - \eta_2\right)\right\|_2^2. 
    \end{align*}
    Define a $\delta$-packing of $\mathcal{B}_0(s_\eta)$ in metric $\rho$ as $\left\{\eta_1,\cdots,\eta_M\right\}\subseteq \mathcal{B}_0(s_\eta)$ such that $\rho(\eta_i,\eta_j) >\delta$ for all $i\not= j$. We construct a packing of $\mathcal{B}_0(s_\eta)$ as $\mathcal{H}_{s_\eta}$ in which each $\eta\in \mathcal{H}_{s_\eta}$ satisfies $\|\eta\|_0 = s_\eta$ and any nonzero element of $\eta$ is $\gamma \not = 0$. From the Varshamov-Gilbert bound, there exist such a $\mathcal{H}_{s_\eta}$ with 
    \begin{align*}
        \quad \left|\mathcal{H}_{s_\eta}\right| \ge \exp\left\{\frac{s_\eta}{2}\log \frac{p - s_\eta}{s_\eta/2}\right\}
    \end{align*}\
    satisfies that for any $\eta_i,\eta_j\in \mathcal{H}_{s_\eta}$,
    \begin{align*}
        \sum_{l=1}^p \mathbf{1}_{(\eta_i)_l \not = (\eta_j)_l} \ge \frac{s_\eta}{2}.
    \end{align*}
    
    Hence $\|\eta_i-\eta_j\|_2 \ge \gamma \sqrt{s_\eta/2}$. Choose $\gamma = \delta\sqrt{2/s_\eta}$, we know that such $\mathcal{H}_{s_\eta}$ is a $\delta$-packing of $\mathcal{B}_0(s_\eta)$ in $l_2$ norm. 
    
    So
    \begin{align*}
        D_{KL}\left(P_t(\eta_i)\|P_t(\eta_j)\right) = \frac{1}{2\sigma^2} \left\|X_t\left(\eta_i - \eta_j\right)\right\|_2^2\le\frac{2 n_t \|\eta_i-\eta_j\|_2^2}{\sigma^2} \le\frac{2 n_t s_\eta\delta^2}{\sigma^2}.
    \end{align*}
    With Fano's lemma, we know that 
    \begin{align*}
        \inf_{\hat{\eta}} \sup_{\eta \in \mathcal{B}_0(s_\eta)} \mathbb{P}\left(\|\hat{\eta} - \eta_0\|_2 \ge \delta\right) & \ge 1- \frac{\max_{i,j}D_{KL}\left(P_t(\eta_i)\|P_t(\eta_j)\right) +\log 2}{\log \left|\mathcal{H}_{s_\eta}\right|}\\ &\ge
        1- \frac{2(2n_t \delta^2 +\log 2)}{s_\eta \log (2(p-s_\eta)/s_\eta)} \ge \frac{1}{2},
    \end{align*}
    if we choose $\delta = c_0\sqrt{s_\eta \log p/n_t}$ for a small enough constant $c_0$. 
    
    Similarly, we have $\|\eta_i-\eta_j\|_1 \ge \gamma s_\eta/2$. Choose $\gamma = 2\delta/s_\eta $, we know that such $\mathcal{H}_{s_\eta}$ is a $\delta$-packing of $\mathcal{B}_0(s_\eta)$ in $l_1$ norm. 

    So
    \begin{align*}
        D_{KL}\left(P_t(\eta_i)\|P_t(\eta_j)\right)  \le \frac{4 n_t \|\eta_i-\eta_j\|_1^2}{\sigma^2} \le\frac{4 n_t \delta^2}{\sigma^2},
    \end{align*}
    this is because $\|\eta_i-\eta_j\|_2 \le \|\eta_i-\eta_j\|_1$. 
    With Fano's lemma, we know that 
    \begin{align*}
        \inf_{\hat{\eta}} \sup_{\eta \in \mathcal{B}_0(s_\eta)} \mathbb{P}\left(\|\hat{\eta} - \eta_0\|_1 \ge \delta\right) & \ge 1- \frac{\max_{i,j}D_{KL}\left(P_t(\eta_i)\|P_t(\eta_j)\right) +\log 2}{\log \left|\mathcal{H}_{s_\eta}\right|}\\ &\ge
        1- \frac{2(4n_t  \delta^2 +\log 2)}{s_\eta \log (2(p-s_\eta)/s_\eta)} \ge \frac{1}{2},
    \end{align*}
    if we choose $\delta = c_0\sqrt{s_\eta \log p/n_t}$. 
\end{pf}

    in event \ref{set:g_11}. So
    \begin{align}
       & \left|\frac{2}{n_s}\left(\beta_s - \hat{\beta}_s\right)^\top\left(\sum_{k=1}^K \tilde{X}_s^{(k)\top}\tilde{H}_s^{(k)}\phi_s^{(k)}\right)\right|\notag \\  \le\ &  \left\|\frac{2}{n_s}\tilde{X}_s b_s\right\|_2\left\|\tilde{X}_s\left(\beta_s - \hat{\beta}_s\right)\right\|_2+2\left(C_kC_\phi \sqrt{\frac{q\log p}{n_s}}+  2C_t \sqrt{\frac{p}{n_s}}\lambda_\Psi^{-1} \|\phi_t\|_2\right) \left\|\beta_s - \hat{\beta}_s\right\|_1\label{eq:error_bound_B6}. 
    \end{align}

    \begin{align}
        & \left|\frac{2}{n_s}\left(\beta_s - \hat{\beta}_s\right)^\top \left(\sum_{k=1}^K \tilde{X}_s^{(k)\top}\tilde{X}_s^{(k)}\left(\beta_s^{(k)}-\frac{1}{n_s}\sum_{k=1}^K n_s^{(k)}\beta_s^{(k)}\right)\right)\right| \notag\\ \le\ &

    in event \ref{set:g_12}. 

    With the bounds \eqref{eq:error_bound_B5},\eqref{eq:error_bound_B6} and \eqref{eq:error_bound_B7} or error terms \ref{term:B_5},\ref{term:B_6} and \ref{term:B_7}, it holds that
    \begin{align*}
         \frac{1}{n_s}\left\|\tilde{X}_s\left(\hat{\beta}_s - \beta_s\right)\right\|_2^2 &\le \lambda_s\left(\|\beta_s\|_1 - \|\hat{\beta}_s\|_1\right) \\&+ 2\left(\left(C_\varepsilon +C_\delta C_x+C_kC_\phi\right)\sqrt{\frac{\log p}{n_s}}+   2C_t \sqrt{\frac{p}{n_s}}\lambda_\Psi^{-1} \|\phi_t\|_2\right)\left\|\beta_s - \hat{\beta}_s\right\|_1\\&+\left\|\frac{2}{n_s}\tilde{X}_s b_s\right\|_2\left\|\tilde{X}_s\left(\beta_s - \hat{\beta}_s\right)\right\|_2.

            \|\hat{\beta}_s - \beta_s\|_2 &\le \left(\frac{3}{2C_r}\lambda_s \|(\hat{\beta}_s - \beta_s)_{S_s}\|_1\right)^{1/2} \le \frac{3}{2C_t}\sqrt{s}\lambda_s. 
            \end{aligned}
        \end{equation}

\textbf{Case 2.} If \begin{align*}
            \left\|\frac{2}{n_s}\tilde{X}_s b_s\right\|_2\left\|\tilde{X}_s\left(\beta_s - \hat{\beta}_s\right)\right\|_2 \ge \frac{\lambda_s}{4}\left\|(\hat{\beta}_s - \beta_s)_{S_s}\right\|_1.

            \|(\hat{\beta}_s - \beta_s)_{S_s}\|_1 \le \frac{8}{\lambda_s}\left\|\frac{1}{n_s}\tilde{X}_s b_s\right\|_2\left\|\tilde{X}_s\left(\beta_s - \hat{\beta}_s\right)\right\|_2.

        \end{align}
        So
        \begin{align*}
            \frac{1}{n_s}\left\|\tilde{X}_s\left(\hat{\beta}_s - \beta_s\right)\right\|_2 \le 10 \left\|\frac{1}{n_s}\tilde{X}_s b_s\right\|_2. 
        \end{align*}
        The inequality \eqref{eq:confounding_dominate} also gives us that

            \|(\hat{\beta}_s - \beta_s)_{S_s^c}\|_1 \le \frac{40}{\lambda_s}\left\|\frac{1}{n_s}\tilde{X}_s b_s\right\|_2\left\|\tilde{X}_s\left(\beta_s - \hat{\beta}_s\right)\right\|_2. 
        \end{align*}
        So
        \begin{align}
        \label{eq:case2_error}
            \|\hat{\beta}_s - \beta_s\|_2 \le \|\hat{\beta}_s - \beta_s\|_1 &\le  \|(\hat{\beta}_s - \beta_s)_{S_s}\|_1 + \|(\hat{\beta}_s - \beta_s)_{S_s^c}\|_1\notag \\&   \le \frac{48}{\lambda_s}\left\|\frac{1}{n_s}\tilde{X}_s b_s\right\|_2\left\|\tilde{X}_s\left(\beta_s - \hat{\beta}_s\right)\right\|_2\notag\\& \le
            \frac{480}{\lambda_s}\left\|\frac{1}{\sqrt{n_s}}\tilde{X}_s b_s\right\|_2^2
        \end{align}

        With \eqref{eq:case1_error} and \eqref{eq:case2_error}, it holds for a constant $C$ that
        \begin{align*}
            \|\hat{\beta}_s - \beta_s\|_2 & \le \frac{C}{C_r}\sqrt{s}\lambda_s + \frac{1}{\lambda_s} \left\|\frac{1}{\sqrt{n_s}}\tilde{X}_s b_s\right\|_2^2,\\
            \|\hat{\beta}_s - \beta_s\|_1 & \le \frac{C}{C_r} s\lambda_s + \frac{1}{\lambda_s} \left\|\frac{1}{\sqrt{n_s}}\tilde{X}_s b_s\right\|_2^2. 
        \end{align*}
        Finally, we know that from the definition of the trim transform, 
        \begin{align*}
            \left\|\frac{1}{\sqrt{n_s}}\tilde{X}_s b_s\right\|^2_2 \le C\frac{p}{n_s}\|\Sigma_X^{-1} \mathbb{E}[f(H)^\top H]\phi_t\|^2_2 \le C \frac{p\|\phi_t\|_2^2}{n_s \lambda_\Psi^2},
        \end{align*}
        with Assumption 2. 
\end{pf}

\begin{lemma}
    \label{lem:trimed_noise_error}
    If $E_i,H_i,i=1,\cdots,n$ are i.i.d. sampled from distributions for $E$ and $H$ that satisfies Assumption 1 and the function $f$ satisfies the condition for $f$ in Assumption 1, then with probability larger than $1 - p^{-c}$,
    \begin{align*}
        \left\|\frac{1}{n}\tilde{X}_s^{(k)\top}\tilde{\varepsilon}_s^{(k)} \right\|_{\infty}\le C_\varepsilon\sqrt{\frac{\log p}{n_s}}. 
    \end{align*}
    for some strictly positive constants $c, C_\varepsilon$. 
\end{lemma}

\begin{pf}{\it of Lemma \ref{lem:trimed_noise_error}. }
    Because that $\varepsilon_s$ is independent to $X_s$ and $\varepsilon_{s,\cdot,i},1\le i\le n_s$ are mean zero sub-Gaussian random variables. We have 
    \begin{align*}
        \mathbb{P}\left(\left\|\frac{1}{n}\tilde{X}_s^{(k)\top}\tilde{\varepsilon}_s^{(k)} \right\|_{\infty}\ge C_\varepsilon\sqrt{\frac{\log p}{n_s}}\right) \le \sum_{i=1}^p \mathbb{P}\left(\left|\frac{1}{n}\sum_{j=1}^{n_s}\tilde{X}_{s,i,j}^{(k)\top}\tilde{\varepsilon}_{s,,j}^{(k)} \right|\ge C_\varepsilon\sqrt{\frac{\log p}{n_s}}\right) \le p^{-c}. 
    \end{align*}
\end{pf}

    If $E_i,H_i,i=1,\cdots,n$ are i.i.d. sampled from distributions for $E$ and $H$ that satisfies Assumption 1 and the function $f$ satisfies the condition for $f$ in Assumption 1, then with probability larger than $1 -  p^{-c}$,
    \begin{align*}
        \left\|\frac{1}{n_s}\mathcal{V}^2_s \sum_{k=1}^K X_s^{(k)\top}X_s^{(k)}\left(\beta_s^{(k)}-\frac{1}{n_s}\sum_{k=1}^K n_s^{(k)}\beta_s^{(k)}\right)\right\|_{\infty} \le  C_ks_\delta \sqrt{\frac{\log p}{n_s}},
    \end{align*}
    for some strictly positive constants $c,C_k$. 
\end{lemma}
\begin{pf}{\it of Lemma \ref{lem:trimed_sample_covariance}. }
From Assumption 1, we know that $E_{s,\cdot,i}$, $H_{s,\cdot,i}$, $f(H_s)_{\cdot,i}$ and $\varepsilon_{\cdot}$ follows sub-Gaussian distributions. So $X_s^\top X_s/n_s$ is a random matrix with $(X_{s})_{i,\cdot}$ as i.i.d. sub-Gaussian random variables. 

Consider the perturbation form: $X_s^\top X_s/n_s = \Sigma + \left(X_s^\top X_s/n_s - \Sigma\right):=\Sigma + E$. We know that 
\begin{align*}
    \left\|E\right\|_{op} \le \sqrt{p}\left\|X_s^\top X_s/n_s - \Sigma\right\|_{\infty} \le C_x\sqrt{\frac{p\log p}{n}}
\end{align*}
holds with probability larger than $1-6p^{-c C_x^2/16 +2}$ from Lemma \ref{lem:error_sample_covariance1}. From the Weyl theorem, we know that 
\begin{align*}
    \left|\hat{\lambda}_i - \lambda_i\right| \le \left\|E\right\|_{op} \le C_x\sqrt{\frac{p\log p }{n}}. 
\end{align*}
From the Devis-Kahan theorem in the infinity norm (Theorem 15 in \cite{bhardwaj2024matrix}), if the eigen-gap $\Delta_i = \lambda_i -\lambda_{i+1}$ for those eigenvalues $\lambda_i,1\le i\le \tau$ is larger than $p^{1/2}n^{-1/4}\log^{3/4} p$, then 
\begin{align*}
    \left\|\hat{v}_i - v_i\right\|_{\infty} &\le c\|V\|_{\infty}\left(\|\hat{v}_i - v_i\|_{2} + \frac{\sqrt{\|E\|_{\infty} p}\log p}{\lambda_i}\right)\\& \le c \|V\|_{\infty}\left(\frac{\|E\|_{op}}{\Delta_i}+\frac{\sqrt{\|E\|_{\infty} p}\log p}{\lambda_i}\right)\\& \le 
    c\sqrt{\frac{\log p}{p}} \left(n_s^{-1/4}\log^{-1/4} p +n_s^{-1/4} p^{1/2}\lambda_\tau^{-1} \log^{5/4} p\right) \le c\left(\frac{\log p}{n_sp^2}\right)^{1/4}. 
\end{align*}
Here the second inequality is the Devis-Kahan theorem in the $l_2$ norm, the last inequality using the Assumption 2. 

Define $\mathcal{V} = V\operatorname{diag}(\tau_s/D_{1:\tau},1_{p-\tau})V^\top$ where $\Sigma_X = VDV^\top$. It holds that
\begin{align}
    \left\|\mathcal{V}^2_s \frac{1}{n_s}X_s^{\top}X_s - \mathcal{V}^2\Sigma_X\right\|_{\infty} 
    &\le 
    \left\| X_s^\top X_s - \Sigma_X\right\|_{\infty} + \left\|\left(I_p - \mathcal{V}^2_s\right) X_s^{(k)}X_s - \left(I_p - \mathcal{V}^2\right)\Sigma_X\right\|_{\infty}\notag
    \\&\le  
    \left\| X_s^{(k)}X_s - \Sigma_X\right\|_{\infty} + \sum_{i=1}^\tau \left\|(\hat{\lambda}_i - \tau_s^2)\hat{v}_i\hat{v}^\top_i - (\lambda_i -\tau_s^2)v_iv^\top_i\right\|_{\infty}\notag
    \\&\le 
    C_x\sqrt{\frac{\log p}{n_s}} + \sum_{i=1}^\tau \left(|\hat{\lambda}_i - \lambda_i|\left\|v_{i}\right\|^2_{\infty} +\left|\hat{\lambda}_i-\tau_s^2\right|\left\|\hat{v}_{i} -v_i\right\|^2_{\infty}\right)\notag
    \\&\le 
    C_x\sqrt{\frac{\log p}{n_s}} + C\left(\sum_{i=1}^\tau \sqrt{\frac{p \log p}{n_s}}\left(\sqrt{\frac{\log p}{p}}\right)^2 \right.\notag\\& \left.\hspace{3cm}+ \sum_{i=1}^\tau \left(\sqrt{\frac{p \log p}{n_s}} + |\lambda_i-\tau_s^2|\right)\sqrt{\frac{\log p}{n_s p^2}}\right)\notag
    \\& \le 
     C_k\sqrt{\frac{\log p}{n_s}}\label{eq:trimed_infinity_norm}. 
\end{align}
Here we use Assumption 2 and $\sum_{i=1}^\tau |\lambda_i - \tau_s^2|\le c p$. If the element eigen-gap condition is not hold, we only need the eigen gap $\Delta_\tau = \lambda_\tau - \lambda_{\tau+1}$ is larger than $p^{1/2}n^{-1/4}\log^{3/4} p $ holds for a $\tau$. From Lemma S.13 in supplementary of \cite{chen2024matrix} or Theorem 2 in \cite{pensky2024davis}, we have the subspace version of Devis-kahan theorem in the two-to-infinity norm that 
\begin{align*}
    \left\|\hat{V}_{\cdot,1:\tau} - V_{\cdot,1:\tau}R\right\|_{2,\infty} \le c\|V_{\cdot,1:\tau}\|_{2,\infty}\left(\frac{\|E\|_{op}}{\Delta_\tau} +\frac{\sqrt{\|E\|_{\infty}p}\log p }{\lambda_\tau}\right)
\end{align*}
 holds for a rotation matrix $\mathcal{R}\in \mathbb{R}^{\tau\times \tau}$ and the inequality \eqref{eq:trimed_infinity_norm} still holds. 

So
\begin{align*}
     & \left\|\frac{1}{n_s}\mathcal{V}^2_s \sum_{k=1}^K X_s^{(k)\top}X_s^{(k)}\left(\beta_s^{(k)}-\frac{1}{n_s}\sum_{k=1}^K n_s^{(k)}\beta_s^{(k)}\right)\right\|_{\infty} \\=\ & \left\|\frac{1}{n_s} \sum_{k=1}^K \left(n_s^{(k)}\mathcal{V}^2_s\left(\frac{1}{n_s^{(k)}}X_s^{(k)\top}X_s^{(k)}-\mathcal{V}^2\Sigma_X\right)\right)\left(\beta_s^{(k)}-\frac{1}{n_s}\sum_{k=1}^K n_s^{(k)}\beta_s^{(k)}\right)\right\|_{\infty}
     \\\le \ &
     \left\|\frac{1}{n_s}\mathcal{V}^2_s \sum_{k=1}^K X_s^{(k)\top}X_s^{(k)} - \mathcal{V}^2\Sigma_X\right\|_{\infty}\sum_{k=1}^K \frac{n_s^{(k)}}{n_s}\left\|\beta_s^{(k)}-\frac{1}{n_s}\sum_{k=1}^K n_s^{(k)}\beta_s^{(k)}\right\|_{1}\\\le \ &
      C_k s_\delta\sqrt{\frac{\log p}{n_s}}. 
\end{align*}
\end{pf}

\subsection{Proof of Proposition 3}
\begin{pf}{\it of Proposition 3. }
    We consider the special case that $\beta_s^{(k)} = \beta_s,1\le k\le K$ and the noise distribution is Gaussian distribution $N(0,\sigma^2)$. We then consider the case that $f(H)= H\Psi$ with $\lambda_1(\Psi) = \cdots = \lambda_q(\Psi) = \lambda_\Psi$. 

    Define the distribution of target data as $P_t(\beta_t,\phi_t)$ and the distribution of source data as $P_s(\beta_s,\phi_t)$. Define that 
    \begin{align*}
        D_{KL}(P(\beta_{t1})\|P(\beta_{t2})) = \underset{\substack{\beta_{t1}=\beta_1+\eta_1,\beta_{t2}=\beta_2+\eta_2\\\beta_1,\beta_2\in \mathcal{B}_0(s),\eta_1,\eta_2\in \mathcal{B}_0(s_\eta)\\\phi_1,\phi_2\in\mathcal{B}_2(c_\phi)}}{\min} &D_{KL} (P_t(\beta_1+\eta_1,\phi_1)\|P_t(\beta_2+\eta_2,\phi_2)) \\+& D_{KL} (P_s(\beta_1,\phi_1)\|P_s(\beta_2,\phi_2)). 
    \end{align*}
Similar to those in the proof of Proposition 1, we define the $\delta$-packing of $\mathcal{B}_0(s_\eta+s)$ as $\mathcal{H}_{s_\eta+s}$ in $l_2$ norm. So 
    \begin{align*}
        \left|\mathcal{H}_{s_\eta}\right|\ge\exp\left\{\frac{s_\eta+s}{2}\log \frac{p-s_\eta-s}{(s_\eta+s)/2}\right\}.
    \end{align*}
    From Fano's lemma,
    \begin{align*}
        \inf_{\hat{\beta}_t}\underset{\substack{\beta_t=\beta+\eta\\\eta\in \mathcal{B}_0(s_\eta),\beta\in \mathcal{B}_0(s)}}{\sup}\mathbb{P}(\|\hat{\beta}_t-\beta_t\|_2 \ge \delta) \ge 1- \frac{\max_{i,j}D_{KL}(P(\beta_{ti})\|P(\beta_{tj})) +\log 2}{\log |\mathcal{H}_{s_\eta+s}|}. 
    \end{align*}
    \textbf{Case 1. }
    We consider $E = 0$ and $H$ follows Gaussian distribution $N(0,I_q)$, then 
    \begin{align*}
        P_t(\beta_t,\phi_t) &= N\left(X_t\beta_t,(\|\phi_t\|_2^2 + \sigma^2)I_{n_t}\right),\\
        P_s(\beta_s,\phi_t) &= N\left(X_s\beta_s,(\|\phi_t\|_2^2 + \sigma^2)I_{n_s}\right).
    \end{align*}
    So for any $\eta_1,\eta_2 \in \mathcal{B}_0((s_\eta)$, $\beta_1,\beta_2\in \mathcal{B}_0(s)$ and $\phi_1,\phi_2 \in \mathcal{B}_2(c_\phi)$, we have 
    \begin{align*}
        & D_{KL} (P_t(\beta_1+\eta_1,\phi_1)\|P_t(\beta_2+\eta_2,\phi_2)) \\= \ & 
        \frac{n_t}{2}\log \frac{\|\phi_2\|_2^2 + \sigma^2}{\|\phi_1\|_2^2 + \sigma^2} + \frac{\|X_t(\eta_1-\eta_2 + (\beta_1-\beta_2))\|_2^2 + n_t(\|\phi_1\|_2^2 - \|\phi_2\|_2^2)}{2(\sigma^2 + \|\phi_2\|_2^2)}.
    \end{align*}
    And
    \begin{align*}
        & D_{KL} (P_s(\beta_1,\phi_1)\|P_s(\beta_2,\phi_2)) \\= \ & 
        \frac{n_s}{2}\log \frac{\|\phi_2\|_2^2 + \sigma^2}{\|\phi_1\|_2^2 + \sigma^2} + \frac{\|X_s(\eta_1-\eta_2 + (\beta_1-\beta_2))\|_2^2 + n_s(\|\phi_1\|_2^2 - \|\phi_2\|_2^2)}{2(\sigma^2 + \|\phi_2\|_2^2)}.
    \end{align*}
    So
    \begin{align*}
        D_{KL}(P(\beta_{t1})\|P(\beta_{t2})) \le \underset{\substack{\beta_{t1}=\beta_1+\eta_1,\beta_{t2}=\beta_2+\eta_2,\\\beta_1,\beta_2\in \mathcal{B}_0(s)\\\eta_1,\eta_2\in \mathcal{B}_0(s_\eta)\\
        \|\phi_1\|_2=c_\phi,\phi_2=\phi_1}}{\min} & D_{KL} (P_t(\beta_1+\eta_1,\phi_1)\|P_t(\beta_2+\eta_2,\phi_2))\\ + & D_{KL} (P_s(\beta_1,\phi_1)\|P_s(\beta_2,\phi_2))\\
        &\hspace{-4cm}=  \frac{\|X_t((\eta_1-\eta_2)+(\beta_1-\beta_2))\|_2^2 + \|X_s(\beta_1-\beta_2)\|_2^2 }{2(\sigma^2 + \|\phi_2\|_2^2)}\\& \hspace{-4cm}\le
        \frac{C\lambda_\Psi^2 p^{-1} (n_t\|\eta_1-\eta_2 + (\beta_1-\beta_2)\|_2^2 + n_s\|\beta_1-\beta_2\|_2^2)}{(2(\sigma^2 + \|\phi_2\|_2^2)}.
    \end{align*}
    Choose $\delta = c_1p^{1/2} n_s^{-1/2}\lambda_\Psi^{-1}C_\phi$, with $n_s\gg n_t$, then
    \begin{align*}
        D_{KL}(P(\beta_{t1})\|P(\beta_{t2})) \le 3. 
    \end{align*}
    So
    \begin{align*}
         \inf_{\hat{\beta}_t}\underset{\substack{\beta_t=\beta+\eta\\\eta\in \mathcal{B}_0(s_\eta),\beta\in \mathcal{B}_0(s)}}{\sup}\mathbb{P}\left(\|\hat{\beta}_t-\beta_t\|_2 \ge c_1\sqrt{\frac{p\|\phi_t\|_2^2}{n_s\lambda_\Psi^2}}\right) \ge 1- \frac{6+ 2\log 2}{(s_\eta+s)\log(2(p-s_\eta-s)/(s_\eta+s)} \ge \frac{1}{2}. 
    \end{align*}
    \textbf{Case 2.} We consider $H=0,c_\phi=0$ and $E$ follows Gaussian distribution $N(0,I_p)$. Under these conditions, the problem reduces to transfer learning for high-dimensional linear regression. Theorem B in the supplementary materials of \cite{li2022transfer} provides the minimax lower bound as follows:
    \begin{align*}
        \inf_{\hat{\beta}_t}\underset{\substack{\beta_t=\beta+\eta\\\eta\in \mathcal{B}_0(s_\eta),\beta\in \mathcal{B}_0(s)}}{\sup}\mathbb{P}\left(\|\hat{\beta}_t-\beta_t\|_2 \ge c_2\sqrt{\frac{s_\eta \log p}{n_t}} \vee \sqrt{\frac{s\log p}{n_s}}\right) \ge  \frac{1}{2}. 
    \end{align*}
    With the assumption that $n_s\gg n_t$ and $s_\eta,s$ are constants, finally we have
    \begin{align*}
        \inf_{\hat{\beta}_t}\underset{\substack{\beta_t=\beta+\eta\\\eta\in \mathcal{B}_0(s_\eta),\beta\in \mathcal{B}_0(s)}}{\sup}\mathbb{P}\left(\|\hat{\beta}_t-\beta_t\|_2 \ge c_1\sqrt{\frac{p\|\phi_t\|_2^2}{n_s\lambda_\Psi^2}}\vee c_2\sqrt{\frac{s_\eta \log p}{n_t}} \right) \ge  \frac{1}{2}. 
    \end{align*}
\end{pf}


\subsection{Proof of Proposition 2 and 4}

    We consider the special case that $\beta_s^{(k)} = \beta_s, 1\le k \le K$ and the noise distribution is Gaussian distribution $N(0,\sigma^2)$. We then consider the case that $f(H) = H\Psi$ with $\lambda_1(\Psi) = \cdots = \lambda_q(\Psi) = \lambda_\Psi$. 
    
    Define the distribution of target data as $P_t(\eta,\beta_s,\phi_t)$. Also define that

    \end{align*}
    From Fano's lemma,
    \begin{align*}
        \inf_{\tilde{\eta}}\sup
        _{\eta\in \mathcal{B}_0(s_\eta)}\mathbb{P}(\|\tilde{\eta}-\eta_0\|_2 \ge \delta) \ge 1- \frac{\max_{i,j}D_{KL}(P_t(\eta_i)\|P_t(\eta_j)) +\log 2}{\log\|\mathcal{H}_{s_\eta}\|}.

    \textbf{Case 1. }
    We consider $E$ = 0 and $H$ follows Gaussian distribution $N(0,I_q)$, then
    \begin{align*}
        P_t(\eta,\beta_s,\phi_t) = N\left(X_t\left(\eta +\beta_s - \hat{\beta}_s\right), \left(\|\phi_t\|_2^2 + \sigma^2\right) I_n\right).

    So for $\eta_1,\eta_2\in \mathcal{B}_0(s_\eta)$, $\beta_1,\beta_2\in \mathcal{B}_0(s)$ and $\phi_1,\phi_2 \in \mathcal{B}_2(c_\phi)$, we have 
    \begin{align*}
        & D_{KL}(P_t(\eta_1,\beta_1,\phi_1)\|P_t(\eta_2,\beta_2,\phi_2))\\ =\ & \frac{n_t}{2} \log \frac{\|\phi_2\|_2^2 +\sigma^2}{\|\phi_1\|_2^2 + \sigma^2} + \frac{\|X_t(\eta_1-\eta_2 + (\beta_1 -\hat{\beta}_1) - (\beta_2 -\hat{\beta}_2))\|_2^2 + n_t(\|\phi_1\|_2^2 - \|\phi_2\|_2^2)}{2(\sigma^2 + \|\phi_2\|_2^2)}. 
    \end{align*}
   
    So
    \begin{align*}
        D_{KL}(P_t(\eta_1)\|P_t(\eta_2)) &\le \min_{\beta_1,\beta_2\in \mathcal{B}_0(s),\|\phi_1\|_2=c_\phi,\phi_2=\phi_1} D_{KL}(P_t(\eta_1,\beta_1,\phi_1)\|P_t(\eta_2,\beta_2,\phi_2))\\&= 
        % \log \frac{4\|\phi_1\|_2^2 +\sigma^2}{\|\phi_1\|_2 +\sigma^2} + \frac{\|X_t(\eta_1-\eta_2 + (\beta_1 -\hat{\beta}_1) - (\beta_2 -\hat{\beta}_2))\|_2^2 - 3\|\phi_1\|_2^2}{2(\sigma^2 + 4\|\phi_1\|_2^2)}\\&\le 
        % \frac{3\|\phi_1\|_2^2}{\|\phi_1\|_2^2 +\sigma^2} + \frac{C\lambda_\Psi^2 p^{-1}n_t\|\eta_1-\eta_2+ (\beta_1 -\hat{\beta}_1) - (\beta_2 -\hat{\beta}_2)\|^2_2- 3\|\phi_1\|_2^2}{2(\sigma^2 + 4\|\phi_1\|_2^2)}. 
         \frac{C\lambda_\Psi^2 p^{-1}n_t\|\eta_1-\eta_2+ (\beta_1 -\hat{\beta}_1) - (\beta_2 -\hat{\beta}_2)\|^2_2}{2(\sigma^2 + \|\phi_1\|_2^2)}. 
    \end{align*}
    Here the last inequality is from $X_t = H_t\Psi$ and $\eta_1-\eta_2+ (\beta_1 -\hat{\beta}_1) - (\beta_2 -\hat{\beta}_2))$ is independent to $H_t\Psi$. So with probability larger than $1-p^{-c}$, the length of the projection for the direction vector of $\eta_1-\eta_2+ (\beta_1 -\hat{\beta}_1) - (\beta_2 -\hat{\beta}_2)$ to the row space of $\Psi$ can be bounded by $C p^{-1/2}$. 

    Choose $\delta = c_1 p^{1/2}n_t^{-1/2}\lambda_\Psi^{-1}c_\phi$ for some constant $c_1$, then
    \begin{align*}
        &\frac{C\lambda_\Psi^2 p^{-1}n_t\|\eta_1-\eta_2+ (\beta_1 -\hat{\beta}_1) - (\beta_2 -\hat{\beta}_2)\|^2_2}{2(\sigma^2 + \|\phi_1\|_2^2)} \\\le \ &
        \frac{2C\lambda_\Psi^2 p^{-1} n_t\left(\delta^2 +  (C\sqrt{s}pc_\phi^2)/(n_s\lambda_\Psi^2)\right)}{2(\sigma^2 + \|\phi_1\|_2^2)}
        \le 3. 
    \end{align*}
    The first inequality uses Theorem 2 and $n_s\gg n_t$. Then
    \begin{align*}
        D_{KL}(P_t(\eta_1)\|P_t(\eta_2)) \le 3.
    \end{align*}
    So
    \begin{align}
         \inf_{\tilde{\eta}}\sup
        _{\eta\in \mathcal{B}_0(s_\eta)}\mathbb{P}\left(\|\tilde{\eta}-\eta_0\|_2 \ge c_1 \sqrt{\frac{p\|\phi_t\|_2^2}{n_t\lambda_\Psi^2}}\right) \ge 1- \frac{6 +2\log 2}{s_\eta\log (2(p-s_\eta)/s_\eta)} \ge \frac{1}{2}.\label{eq:lower_bound_case1}
    \end{align}
    \textbf{Case 2. }
    We consider $H=0,c_\phi=0$ and $E$ follows Gaussian distribution $N(0,I_p)$, then
    \begin{align*}
        D_{KL}(P_t(\eta_1)\|P_t(\eta_2)) & \le \frac{1}{2\sigma^2}\|X_t(\eta_1-\eta_2 + (\beta_1 -\hat{\beta}_1) - (\beta_2 -\hat{\beta}_2))\|_2^2 \\& \le 
        \frac{2n_t \|\eta_i-\eta_j + (\beta_1 -\hat{\beta}_1) - (\beta_2 -\hat{\beta}_2)\|_2^2}{\sigma^2}. 
    \end{align*}
    Choose $\delta = c_2\sqrt{s_\eta \log p/n_t}$, with $\|\beta_s - \hat{\beta}_s\|_2 \le c\sqrt{s\log p/n_s}$, we know that 
    \begin{align*}
        D_{KL}(P_t(\eta_1)\|P_t(\eta_2)) \le \frac{c_2s_\eta \log p}{\sigma^2}. 
    \end{align*}
    and then 
    \begin{align}
        \inf_{\tilde{\eta}}\sup
        _{\eta\in \mathcal{B}_0(s_\eta)}\mathbb{P}\left(\|\tilde{\eta}-\eta_0\|_2 \ge c_2\sqrt{\frac{s_\eta\log p}{n_t}}\right) \ge \frac{1}{2}.\label{eq:lower_bound_case2}
    \end{align}

    With \eqref{eq:lower_bound_case1} and \eqref{eq:lower_bound_case2}, we have
    \begin{align*}
         \inf_{\tilde{\eta}}\sup
        _{\eta\in \mathcal{B}_0(s_\eta)}\mathbb{P}\left(\|\tilde{\eta}-\eta_0\|_2 \ge c_1 \sqrt{\frac{p\|\phi_t\|_2^2}{n_t\lambda_\Psi^2}} \vee c_2\sqrt{\frac{s_\eta\log p}{n_t}}\right) \ge \frac{1}{2}.
    \end{align*}

    For the $l_1$ norm lower bound. The case 1 is the same as above proof. For case 2, we only need to choose $\delta = c_2\sqrt{s_\eta \log p/n_t}$. Because $\|\eta_i-\eta_j\|_2 \le  \|\eta_i-\eta_j\|_1$. So we still have 
    \begin{align*}
        D_{KL}(P_t(\eta_1)\|P_t(\eta_2)) \le \frac{c_2s_\eta \log p}{\sigma^2}. 
    \end{align*}
    So
    \begin{align*}
        \inf_{\tilde{\eta}}\sup
        _{\eta\in \mathcal{B}_0(s_\eta)}\mathbb{P}\left(\|\tilde{\eta}-\eta_0\|_1 \ge c_1 \sqrt{\frac{p\|\phi_t\|_2^2}{n_t\lambda_\Psi^2}} \vee c_2\sqrt{s_\eta\frac{\log p}{n_t}}\right) \ge \frac{1}{2}.
    \end{align*}

    For the definition of $\tilde{\beta}_t = \tilde{\eta} + \tilde{\beta}_s$, we have
    \begin{align*}
        \mathbb{P}\left(\|\tilde{\beta}_t - \beta_t\|_2 \ge \delta \right) \ge \mathbb{P} \left(\|\tilde{\eta} - \eta_0\|_2 - \|\tilde{\beta}_s - \beta_s\|_2 \ge \delta \right) \ge \mathbb{P}\left(\|\tilde{\eta} - \eta_0\|_2\ge \frac{\delta}{2}\right). 
    \end{align*}
    This is because $\delta \gg \|\tilde{\eta} - \eta_0\|_2$ from $n_s\gg n_t$. 
\end{pf}

    \end{align*}
\end{pf}


\subsection{Lemmas for Theorem 4. }
\begin{lemma}
    \label{lem:l1_bound_etak}
    Under the same conditions as in Theorem 4, with probability larger than $1-p^{-c}$, 
    \begin{align*}
        \left\|\hat{\eta}^{(k)} - \left(\beta_t - \beta_s^{(k)}\right)\right\|_1 &\le
         C s_\eta\lambda_t^{(k)} +\frac{1}{\lambda_t^{(k)}}\frac{p\|\phi_t - \phi_s^{(k)}\|^2_2}{n_t\lambda_\Psi^2},\\
         \left\|\hat{\eta}^{(k)} - \left(\beta_t - \beta_s^{(k)}\right)\right\|_2 &\le
         C \sqrt{s_\eta}\lambda_t^{(k)} +\frac{1}{\lambda_t^{(k)}}\frac{p\|\phi_t - \phi_s^{(k)}\|^2_2}{n_t\lambda_\Psi^2},
    \end{align*}
    where the tuning parameter $\lambda_t^{(k)}$ satisfies that $\lambda_t^{(k)} \ge (c_1\sqrt{q}\|\phi_s^{(k)}\|+c_2)\sqrt{\log p/n_t}+c_3\lambda_\Psi^{-1}\|\phi_t-\phi_s^{(k)}\|_2$ and $C,c_1,c_2,c_3$ are some constants. 
\end{lemma}
\begin{pf}{\it of Lemma \ref{lem:l1_bound_etak}. }
    Define that $\eta^{(k)} = \beta_t -\beta_s^{(k)},\zeta^{(k)} = \phi_t - \phi_s^{(k)}$. Then define that $b^{(k)} = \Sigma_X^{-1}\mathbb{E}[X^\top H]\zeta^{(k)}$, $\mathcal{V}_t = V_t\operatorname{diag}(\min(\tau_t/D_t),1))V_t^\top$ and $\tilde{\mathcal{V}}_s^{(k)} = V_s^{(k)}\operatorname{diag}(\tau_t/D_{s,1:\tau}^{(k)},1_{p-\tau})V_s^{(k)\top}$. So $\tilde{X}_t = X_t\mathcal{V}_t$ and $\tilde{X}_t^\top\tilde{H}_t =\mathcal{V}_t^2 X_t^\top H_t$. Define several events:
    \begin{align}
      &  \left\{\left\|\left(\mathcal{V}_t^2 \frac{1}{n_t}X_t^\top X_t - \mathcal{V}^2\Sigma_X \right)\left(\beta_s^{(k)}-\hat{\beta}_{\text{init}}^{(k)}\right)\right\|_{\infty} \le C\sqrt{\frac{\log p}{n_t}}\right\}\label{set:g_13}\tag{$\mathcal{G}_{13}$}, \\
     & \left\{\left\|\left(\tilde{\mathcal{V}}_s^{(k)2} \frac{1}{n_s^{(k)}}X_s^{(k)\top} X_s^{(k)} - \mathcal{V}^2\Sigma_X \right)\left(\beta_s^{(k)}-\hat{\beta}_{\text{init}}^{(k)}\right)\right\|_{\infty} \le C\sqrt{\frac{\log p}{n_s^{(k)}}}\right\}\label{set:g_13_k}\tag{$\mathcal{G}_{13}^{(k)}$}, \\
    & \left\{\left\|\left(\mathcal{V}_t^2 \frac{1}{n_t}X_t^\top H_t - \mathcal{V}^2\mathbb{E}[X^\top H] \right)\phi_s^{(k)}\right\|_{\infty} \le C\|\phi_s^{(k)}\|_2\sqrt{\frac{q\log p}{n_t}}\right\}\label{set:g_14}\tag{$\mathcal{G}_{14}$},\\
    & \left\{\left\|\left(\tilde{\mathcal{V}}_s^2 \frac{1}{n_s^{(k)}}X_s^{(k)\top} H_s^{(k)} - \mathcal{V}^2\mathbb{E}[X^\top H] \right)\phi_s^{(k)}\right\|_{\infty} \le C\|\phi_s^{(k)}\|_2\sqrt{\frac{q\log p}{n_s^{(k)}}}\right\}\label{set:g_14_k}\tag{$\mathcal{G}_{14}^{(k)}$},\\
        & \left\{\left\|\mathcal{V}_t^2 \frac{1}{n_t} X_t^\top\varepsilon_t\right\|_{\infty} \le C\sqrt{\frac{\log p}{n_t}}\right\}\label{set:g_15}\tag{$\mathcal{G}_{15}$},\\
    & \left\{\left\|\tilde{\mathcal{V}}_s^{(k)2} \frac{1}{n_s^{(k)}} X_s^{(k)\top}\varepsilon_s^{(k)}\right\|_{\infty} \le C\sqrt{\frac{\log p}{n_s^{(k)}}}\right\}\label{set:g_15_k}\tag{$\mathcal{G}_{15}^{(k)}$},\\
    & \left\{\left\|\frac{1}{n_t}\tilde{X}_t^\top \left(\tilde{H}_t\zeta^{(k)}-\tilde{X}_t b^{(k)}\right)\right\|_{\infty}\le C_t \frac{1}{\lambda_\Psi} \|\zeta^{(k)}\|_2\right\}\label{set:g_16}\tag{$\mathcal{G}_{16}$}. 
    \end{align}
    Define the event $\mathcal{G}^{(k)'} = \mathcal{G}_{13}\cap\mathcal{G}_{13}^{(k)}\cap \mathcal{G}_{14}\cap\mathcal{G}_{14}^{(k)}\cap \mathcal{G}_{15}^{(k)}\cap \mathcal{G}_{16}$. From Lemma \ref{lem:high_probability_set4}, the event $\mathcal{G}^{(k)'}$ happens with a probability 
    \begin{align*}
        \mathbb{P}(\mathcal{G}^{(k)'}) \ge 1- p^{-c}. 
    \end{align*}
     Recall the definition of $\hat{\eta}^{(k)}$. We have the inequality from lasso,
    \begin{align*}
        \frac{1}{n_t}\left\|\tilde{Y}_t - \tilde{X}_t \hat{\beta}_{\text{init}}^{(k)} - \tilde{Z}_t^{(k)} -\tilde{X}_t \hat{\eta}^{(k)}\right\|_2^2 + \lambda_t^{(k)}\|\hat{\eta}^{(k)}\|_1 \le \frac{1}{n_t}\left\|\tilde{Y}_t - \tilde{X}_t \hat{\beta}_{\text{init}}^{(k)} - \tilde{Z}_t^{(k)} - \tilde{X}_t\eta\right\|_2^2 + \lambda_t^{(k)}\|\eta\|_1.  
    \end{align*}
    With $\tilde{Y}_t = \tilde{X}_t\beta_t +\tilde{H}_t\phi_t+ \tilde{\varepsilon}_t = \tilde{X}_t(\beta_s^{(k)} +\eta^{(k)}) + \tilde{H}_t(\phi_s^{(k)} +\zeta^{(k)})+ \tilde{\varepsilon}_t$, we have
    \begin{align*}
        \frac{1}{n_t}\left\|\tilde{X}_t(\hat{\eta}^{(k)} - \eta^{(k)})\right\|_2^2 & \le
        \lambda_t^{(k)}(\|\eta^{(k)}\|_1 - \|\hat{\eta}^{(k)}\|_1) -  \frac{2}{n_t}(\hat{\eta}^{(k)} - \eta^{(k)})^\top\tilde{X}_t^\top \tilde{\varepsilon}_t \\& - \frac{2}{n_t}(\hat{\eta}^{(k)} - \eta^{(k)})^\top \tilde{X}_t^\top \left(\tilde{X}_t\left(\beta_s^{(k)}-\hat{\beta}_{\text{init}}^{(k)}\right) +\tilde{H}_t\phi_t - \tilde{Z}_t^{(k)}\right) \\& \le 
        \lambda_t^{(k)}(\|\eta^{(k)}\|_1 - \|\hat{\eta}^{(k)}\|_1) + 2\|\hat{\eta}^{(k)} - \eta^{(k)}\|_1 \left\|\frac{1}{n_t}\tilde{X}_t^\top \tilde{\varepsilon}_t\right\|_{\infty} \\& +
        2\|\hat{\eta}^{(k)} - \eta^{(k)}\|_1 \left\|\frac{1}{n_t}\tilde{X}_t^\top\left(\tilde{X}_t\left(\beta_s^{(k)}-\hat{\beta}_{\text{init}}^{(k)}\right) +\tilde{H}_t\phi_s^{(k)}- \tilde{Z}_t^{(k)}\right)\right\|_{\infty}\\& + 
        2\|\hat{\eta}^{(k)} - \eta^{(k)}\|_1 \left\|\frac{1}{n_t}\tilde{X}_t^\top \left(\tilde{H}_t\zeta^{(k)}-\tilde{X}_t b^{(k)}\right)\right\|_{\infty}\\& +
        2\left\|\tilde{X}_t(\hat{\eta}^{(k)} - \eta^{(k)})\right\|_2 \left\|\frac{1}{n_t}\tilde{X}_t\zeta^{(k)}\right\|_{2}.
    \end{align*}
    
Consider the term 
    \begin{equation}
    \label{eq:trans_error_for_selection}
        \begin{aligned}
                   &  \left\|\frac{1}{n_t}\tilde{X}_t^\top\left(\tilde{X}_t\left(\beta_s^{(k)}-\hat{\beta}_{\text{init}}^{(k)}\right) +\tilde{H}_t\phi_s^{(k)}- \tilde{Z}_t^{(k)}\right)\right\|_{\infty} \\ \le\ &  \left\|\mathcal{V}_t^2\left[\frac{1}{n_t}X_t^\top\left(X_t\left(\beta_s^{(k)}-\hat{\beta}_{\text{init}}^{(k)}\right) +H_t\phi_s^{(k)}\right)\right] - \tilde{\mathcal{V}}_s^{(k)2}\frac{1}{n_s^{(k)}}X_s^{(k)\top} \hat{Z}_s^{(k)} \right\|_{\infty}\\ +\ &  \left\|\mathcal{V}_t^2\frac{1}{n_t} X_t^\top\hat{Z}_t - \tilde{\mathcal{V}}_s^{(k)2}\frac{1}{n_s^{(k)}}X_s^{(k)\top} \hat{Z}_s^{(k)}\right\|_{\infty}. 
        \end{aligned}
    \end{equation}
    The first term in the right hand of \eqref{eq:trans_error_for_selection} is 
    \begin{align*}
        & \left\|\mathcal{V}_t^2\left[\frac{1}{n_t}X_t^\top\left(X_t\left(\beta_s^{(k)}-\hat{\beta}_{\text{init}}^{(k)}\right) +H_t\phi_s^{(k)}\right)\right] - \tilde{\mathcal{V}}_s^{(k)2}\frac{1}{n_s^{(k)}}X_s^{(k)\top} \hat{Z}_s^{(k)}\right\|_{\infty} \\ =\ & \left\|\left(\mathcal{V}_t^2 \frac{1}{n_t}X_t^\top X_t - \tilde{\mathcal{V}}_s^{(k)2}\frac{1}{n_s^{(k)}}X_s^{(k)\top}X_s^{(k)}\right)\left(\beta_s^{(k)}-\hat{\beta}_{\text{init}}^{(k)}\right)\right\|_{\infty} \\+\ &  \left\|\left(\mathcal{V}_t^2 \frac{1}{n_t}X_t^\top H_t - \tilde{\mathcal{V}}_s^{(k)2}\frac{1}{n_s^{(k)}}X_s^{(k)\top}H_s^{(k)}\right)\phi_s^{(k)}\right\|_{\infty} + \left\|\tilde{\mathcal{V}}_s^{(k)2} \frac{1}{n_s^{(k)}} X_s^{(k)\top}\varepsilon_s^{(k)}\right\|_{\infty},
    \end{align*}
    in which the first term is bounded by \ref{set:g_13} and \ref{set:g_13_k}, the second term is bounded by \ref{set:g_14} and \ref{set:g_14_k}, the third term is bounded by \ref{set:g_15_k}. 
    The second term in the right hand of \eqref{eq:trans_error_for_selection} is 
    \begin{align*}
        \left\|\mathcal{V}_t^2\frac{1}{n_t} X_t^\top\hat{Z}_t - \tilde{\mathcal{V}}_s^{(k)2}\frac{1}{n_s^{(k)}}X_s^{(k)\top} \hat{Z}_s^{(k)}\right\|_{\infty} & \le \left\|\left(\mathcal{V}_t^2\frac{1}{n_t}X_t^\top X_t - \tilde{\mathcal{V}}_s^{(k)2}\frac{1}{n_s^{(k)}}X_s^{(k)\top}X_s^{(k)}\right)\left(\beta_s^{(k)} - \hat{\beta}_{\text{init}}^{(k)}\right)\right\|_{\infty}\\ & +
        \left\|\left(\mathcal{V}_t^2\frac{1}{n_t} X_t^\top H_t-\mathcal{V}_t^2\frac{1}{n_s^{(k)}} X_s^{(k)\top} H_s^{(k)}\right)\phi_s^{(k)} \right\|_{\infty} \\& + 
        \left\|\tilde{\mathcal{V}}_s^{(k)2} \frac{1}{n_s^{(k)}} X_s^{(k)\top}\varepsilon_s^{(k)}\right\|_{\infty}, 
    \end{align*}
    which is similarly bounded in the defined events. So,
    \begin{align*}
         \left\|\frac{1}{n_t}\tilde{X}_t^\top\left(\tilde{X}_t\left(\beta_s^{(k)}-\hat{\beta}_{\text{init}}^{(k)}\right) +\tilde{H}_t\phi_s^{(k)}- \tilde{Z}_t^{(k)}\right)\right\|_{\infty} \le (C_1\sqrt{q}\|\phi_s^{(k)}\|_2 + C_2)\sqrt{\frac{\log p}{n_t}}. 
    \end{align*}

    In event $\mathcal{G}^{(k)'}$, it holds that 
    \begin{align*}
        \frac{1}{n_t}\left\|\tilde{X}_t(\hat{\eta}^{(k)} - \eta^{(k)})\right\|_2^2 & \le
        \lambda_t^{(k)}(\|\eta^{(k)}\|_1 - \|\hat{\eta}^{(k)}\|_1)\\&  + 2\left((C_1\sqrt{q}\|\phi_s^{(k)}\|_2 + C_2)\sqrt{\frac{\log p}{n_t}} +C_t\frac{1}{\lambda_\psi}\|\zeta^{(k)}\|_2\right) \|\hat{\eta}^{(k)} - \eta^{(k)}\|_1 \\& + 
        2\left\|\tilde{X}_t(\hat{\eta}^{(k)} - \eta^{(k)})\right\|_2 \left\|\frac{1}{n_t}\tilde{X}_t\zeta^{(k)}\right\|_{2}.
    \end{align*}
    When $\lambda_t^{(k)}\ge 8 \left((C_1\sqrt{q}\|\phi_s^{(k)}\|_2 + C_2)\sqrt{\log p/n_t} +C_t\lambda_\Psi^{-1}\|\zeta^{(k)}\|_2\right)$, it holds that 
    \begin{align}
    \label{eq:eta_lasso_selection}
        \frac{1}{n_t}\left\|\tilde{X}_t(\hat{\eta}^{(k)} - \eta^{(k)})\right\|_2^2 & \le \lambda_t^{(k)}\|\eta^{(k)}\|_1 - \lambda_t^{(k)}\|\hat{\eta}^{(k)}\|_1 + \frac{\lambda_t^{(k)}}{4}\|\hat{\eta}^{(k)} - \eta^{(k)}\|_1 + \left\|\frac{2}{n_t}\tilde{X}_t\zeta^{(k)}\right\|_{2}\left\|\tilde{X}_t(\hat{\eta}^{(k)} - \eta^{(k)})\right\|_2. 
    \end{align}
    The inequality \eqref{eq:eta_lasso_selection} is similar to \eqref{eq:confouning_lasso_inequality}. Follow the proof of Theorem 2, with Assumption 3.a., it holds that
    \begin{align*}
        \left\|\hat{\eta}^{(k)} - \eta^{(k)}\right\|_1 &\le C_3 s_\eta\lambda_t^{(k)} +\frac{1}{\lambda_t^{(k)}}\frac{p\|\zeta^{(k)}\|^2_2}{n_t\lambda_\Psi^2},\\
        \left\|\hat{\eta}^{(k)} - \eta^{(k)}\right\|_2 &\le C_3 \sqrt{s_\eta}\lambda_t^{(k)} +\frac{1}{\lambda_t^{(k)}}\frac{p\|\zeta^{(k)}\|^2_2}{n_t\lambda_\Psi^2}. 
    \end{align*}
\end{pf}
\begin{lemma}
    \label{lem:high_probability_set4}
    Under Assumption 1, 2 and 3, the event $\mathcal{G}^{(k)'} = \mathcal{G}_{13}\cap\mathcal{G}_{13}^{(k)}\cap \mathcal{G}_{14}\cap\mathcal{G}_{14}^{(k)}\cap \mathcal{G}_{15}^{(k)}\cap \mathcal{G}_{16}$ happens with a probability 
    \begin{align*}
        \mathbb{P}(\mathcal{G}^{(k)'}) \ge 1- p^{-c},
    \end{align*}
    for a constant $c>0$.
\end{lemma}
\begin{pf}{\it of Lemma \ref{lem:high_probability_set4}. }
    The probabilities of events \ref{set:g_13} and \ref{set:g_13_k} are bounded by \eqref{eq:trimed_infinity_norm}. The probabilities of events \ref{set:g_14} and \ref{set:g_14_k} are similarly bounded by the proof of \eqref{eq:trimed_infinity_norm} in Lemma \ref{lem:trimed_sample_covariance}.  The probabilities of events \ref{set:g_15} and \ref{set:g_15_k} are similarly bounded like Lemma \ref{lem:trimed_noise_error}. The probability of event \ref{set:g_16} is bounded similarly to Lemma \ref{lem:trimed_uncorelated_bias}. 
\end{pf}

\section{Theoretical results for the fixed-dimensional scenario}
\subsection{Model shift estimation}
We first give some general assumptions under the fixed-dimensional scenario. 
\begin{assump}
\label{assum:sup:general_assump}
    \begin{enumerate}
    \item[a.] The errors $E_{ti},E_{si}^{(k)}$, the confounders $H_{ti},H_{si}^{(k)}$ and $f(H_{ti}),f(H_{si})^{(k)}$ are i.i.d. sub-Gaussian random variables. 
        \item[b.] The noise terms $\varepsilon_{ti}, \varepsilon_{si}^{(k)}$ are i.i.d. sub-Gaussian random variables with zero mean and variance $\sigma^2$. 
        \item[c.] The smallest eigenvalue of $\operatorname{Cov}(X_{ti})$ is larger than a positive constant $C_r>0$.
        \item[d.] The confounders exert the similar effects on the target and source models, i.e., 
        \begin{align*}
            \left\|\frac{1}{n_s}\sum_{k=1}^K n_s^{(k)}\phi_s^{(k)} - \phi_t\right\|_2 \le C\sqrt{\frac{1}{n}},
        \end{align*}
        where $C$ is a constant. 
    \end{enumerate}
\end{assump}
The Assumption \ref{assum:sup:general_assump}.a. to c. are general assumptions. The Assumption \ref{assum:sup:general_assump}.e. requires that the average confounding effect in the source models is similar to that in the target model, which is similar to Assumption 1.d. under the high-dimensional scenario. 

\begin{thm}
    \label{thm:sup:model_shift}
    Under Assumption \ref{assum:sup:general_assump}, suppose the initial source estimator $\hat{\beta}_{\text{init}}$ for $\beta_s$ satisfies $\|\hat{\beta}_{\text{init}} - \beta_s\|_1 \le C_s$ for a constant $C_s>0$. Then, with probability at least $1-p^{-c}$ for a constant $c>0$, the estimator $\hat{\eta}$ defined in (5) satisfies:
    \begin{align*}
        \|\hat{\eta} - \eta_0\|_1 \lesssim s_\eta \lambda_t,\quad \|\hat{\eta} - \eta_0\|_2 \lesssim \sqrt{s_\eta }\lambda_t,\quad \frac{1}{\sqrt{n_t}}\|X_t(\hat{\eta} - \eta_0)\|_2 \lesssim \sqrt{s_\eta} \lambda_t.
    \end{align*}
    Here the tuning parameter $\lambda_t$ satisfies $\lambda_t\ge (C'_1\sqrt{q}\|\phi_t\|_2+C_2)n_t^{-1/2}$ for some strictly positive constants $C'_1,C'_2$. 
\end{thm}

\begin{pf}{\it of Theorem \ref{thm:sup:model_shift}. }
    Note that now $p$ is fixed which is not increasing with $n$. The logarithm term $\log p$ is now a constant and we still have the bound \eqref{eq:sup:oracle_inequality} that 
    \begin{align*}
       \frac{1}{n_t} \left\|X_t\left(\hat{\eta} - \eta_0\right)\right\|_2^2 \le 
        \lambda_t \|\eta_0\|_1 - \lambda_t \|\hat{\eta}\|_1 + \frac{\lambda_t}{2}\|\hat{\eta} - \eta_0\|_1, 
    \end{align*}
    where we set $\lambda_t \ge C'_1\sqrt{q}\|\phi_t\|_2 +C'_2)n_t^{-1/2}$ for some strictly positive constants $C'_1,C'_2$. With the lower bound $C_r$ of the smallest eigenvalue of $\operatorname{Cov}(X_{ti})$, we know that 
    \begin{align*}
        \frac{1}{n_t}\|X_t(\hat{\eta} - \eta_0)\|_2^2 &\ge C_r \|\hat{\eta} - \eta_0\|_2^2 - \left|(\hat{\eta} - \eta_0)'\left(\Sigma_X - \frac{1}{n_t}X_t'X_t\right)(\hat{\eta} - \eta_0)\right|\\& \gtrsim
        C_r\left(1 - \mathcal{O}_p\left(\sqrt{\frac{p}{n_t}}\right)\right) \|\hat{\eta} - \eta_0\|_2^2 \gtrsim \|\hat{\eta} - \eta_0\|_2^2. 
    \end{align*}
    The remain proof is similar to that of Theorem 1, noting that now we can treat $p,\log p$ as constants. 
\end{pf}

\begin{remark}
    Under the fixed-dimensional setting, the profiled residual transfer (7) can also be replaced by a similar procedure 
    \begin{align}
    \label{eq:sup:profiled_fixed_p}
        \hat{Z}_t^{\text{fix}} = \arg\min_{Z_t\in \mathbb{R}^{n_t}} \left\{\left\|\frac{1}{n_t}X_t^\top Z_t - \frac{1}{n_s}\sum_{k=1}^K X_s^{(k)\top}\hat{Z}_s^{(k)}\right\|_2^2 +\lambda_z \|Z_t\|_2^2 \right\}
    \end{align}
    which employs the $l_2$ norm now for some $\lambda_z>0$. One advantage of $\hat{Z}_t^{\text{fix}}$ is that it has a closed-form expression.  A result analogous to Theorem~\ref{thm:sup:model_shift} can be derived when $\hat{Z}_t^{\text{fix}}$ is used as the estimator of the profiled target residual. 
\end{remark}

\subsection{Target parameter estimation}
Similar to the high-dimensional case, we estimate $\beta_t$ as $\hat{\beta}_t = \hat{\eta} + \hat{\beta}_s$, where $\hat{\beta}_s$ is an estimator of the source parameter $\beta_s$. For simplicity, we focus on the single source scenario $(K=1)$ here and adopt the estimation method proposed by \cite{tang2023synthetic}, which involves a two-step regression procedure. In the first step, the estimator $\hat{\Psi}$ of $\Psi$ is obtained using maximum likelihood estimation or principal component analysis, and a semi-orthogonal matrix $B_{\hat{\Psi}^{\perp}}$ is constructed, whose columns are orthogonal to the columns of $\hat{\Psi}$. Then the synthetic IV $X_s B_{\hat{\Psi}^{\perp}}$ is regressed out to obtain $\hat{X}_s$, and $\beta_s$ is then estimated by 
\begin{align*}
    \hat{\beta}_s = \arg\min_{\beta \in \mathbb{R}^p} \frac{1}{n_s}\left\|Y_s - \hat{X}_s \beta\right\|_2^2,\quad \text{subject to } \|\beta\|_0 \le s,
\end{align*}
where $s$ denotes a tuning parameter controlling sparsity.
\begin{assump}
\label{assum:sup:source_fixed}
    \begin{enumerate}
         \item[a.] The function $f(H) = H\Psi$ for some fixed $\Psi \in \mathbb{R}^{q\times p}$. The errors $E_{ti}, E_{si}$ and the confounders $H_{ti},H_{si}$ are mean zero Gaussian random variables with $\operatorname{Cov}(H_{\cdot i}) = I_q$ and $\operatorname{Cov}(E_{\cdot i}) = D$.
        \item[b.] The true parameter $\beta_s$ has sparsity $s < p - q $.
        \item[c.] There exist an estimator $\hat{\Psi}$ and an orthogonal matrix $O\in \mathbb{R}^{q\times q}$ such that 
        $\|\hat{\Psi} - \Psi O\|_2 \le a_{n_s}$ for some $a_{n_s}\rightarrow 0$ as $n_s\rightarrow \infty$.
    \end{enumerate}
\end{assump}

\begin{thm}
    \label{thm:sup:target_estimation}
    Under Assumptions \ref{assum:sup:general_assump} and \ref{assum:sup:source_fixed}, with probability at least $1-p^{-c}$ for a constant $c>0$, the estimator $\hat{\beta}_t$ satisfies 
    \begin{align*}
       \|\hat{\beta}_t - \beta_t\|_1 \lesssim n_t^{-1/2} \vee a_{n_s} \vee n_s^{-1/2},\quad \|\hat{\beta}_t - \beta_t\|_2 \lesssim n_t^{-1/2} \vee a_{n_s} \vee n_s^{-1/2},
    \end{align*}
    when we set $\lambda_t$ in the model shift estimation as $\lambda_t \asymp n_t^{-1/2}$. 
\end{thm}

\begin{pf}{\it of Theorem \ref{thm:sup:target_estimation}. }
    Theorem 2 in \cite{tang2023synthetic} shows that 
    \begin{align*}
        \|\hat{\beta}_s - \beta_s\|_1 = \mathcal{O}_p(n_s^{-1/2})
    \end{align*}
    when $a_{n_s} \le C n_s^{-1/2}$ for some constant C. When $n_s^{1/2}a_{n_s}\rightarrow \infty$, we can use only $a_{n_s}^{-2}$ data in the source dataset. Still from Theorem 2 in \cite{tang2023synthetic}, we have
    \begin{align*}
        \|\hat{\beta}_s - \beta_s\|_1 = \mathcal{O}_p(a_{n_s}). 
    \end{align*}
    Combine this with Theorem \ref{thm:sup:model_shift} and set $\lambda_t \asymp n_t^{-1/2}$, we have 
    \begin{align*}
        \|\hat{\beta}_t - \beta_t\|_1 \lesssim n_t^{-1/2} \vee a_{n_s} \vee n_s^{-1/2}.
    \end{align*}
    Due to the fixed dimensional $p$, this also holds for $l_2$ norm. 
\end{pf}

\begin{remark}
    When directly using the target data or applying classical transfer learning methods, the available upper bound is $n_t^{-1/2}\vee a_{n_t} \vee n_s^{-1/2}$. Compared with our approach, when the convergence of $\hat{\Psi}$ is slow, these methods are constrained by the slower rate $a_{n_t}$. In contrast, our method can still achieve the optimal rate of $n_t^{-1/2}$ once $n_s$ is sufficiently large. This finding is consistent with results under the high-dimensional setting, further demonstrating the generality of the proposed profiled transfer learning framework.
\end{remark}


\section{Implementation Details}
In this section, we describe the estimation and implementation procedures. For the source estimator $\hat{\beta}_s$, we adopt the standard lasso with cross-fitting as implemented in the R package \textit{glmnet} in the high-dimensional setting. In the fixed-$p$ regime, we follow the approach proposed in \cite{tang2023synthetic}. The number of confounders is estimated via maximum likelihood under a Gaussian assumption, and the estimator $\hat{\beta}_s$ is obtained through a model selection procedure with $l_0$ constraints.

For the model shift estimator $\hat{\eta}$, we again apply the standard lasso to the pseudo-response $Y_t - X_t \hat{\beta}_{\text{init}} - \hat{Z}_t$. To implement the profiled residual transfer, the optimization problem
\begin{align*}
    \hat{Z}_t = \argmin_{Z_t \in \mathbb{R}^{n_t}}\left\{\left\|\frac{1}{n_t}X_t^\top Z_t - \frac{1}{n_s}\sum_{k=1}^K X_s^{(k)\top}\hat{Z}_s^{(k)}\right\|_{\infty}\right\}
\end{align*}
can be reformulated as a linear program:
\begin{align*}
    \hat{z} = \argmin_{0\preceq z\in \mathbb{R}^{2n_t+1}} \alpha^\top z,\quad \operatorname{s.t.} Az\le b,
\end{align*}
where 
\begin{align*}
   \alpha = \left(\begin{matrix}0_{2n_t}\\1\\\end{matrix}\right),\quad A = \left(\begin{matrix}\frac{1}{n_t}X_t^\top&-\frac{1}{n_t}X_t^\top&-1\\-\frac{1}{n_t}X_t^\top&\frac{1}{n_t}X_t^\top&-1\\\end{matrix}\right),\quad b = \left(\begin{matrix}\frac{1}{n_s}\sum_{k=1}^K X_s^{(k)\top}\hat{Z}_s^{(k)}\\-\frac{1}{n_s}\sum_{k=1}^K X_s^{(k)\top}\hat{Z}_s^{(k)}\\\end{matrix}\right).
\end{align*}
This problem can be efficiently solved using interior-point algorithms, yielding 
$$\hat{Z}_t =\hat{z}_{[1:n_t]} - \hat{z}_{[n_t+1:2n_t]}.$$

\section{Additional Simulation}

\subsection{Fixed-dimensional scenario}
In parallel with the high-dimensional setting discussed in the main text, we conduct additional simulations for a fixed-dimensional scenario. We consider the linear confounding structure $X_l = H_l^\top \Psi + E_l$ for $l=s,t$ and generate the source and target data $(X_{li},Y_{li})$ using the same mechanism as in the main text. The covariate dimension is fixed at $p=10$, which is much smaller than the sample size. 

In the single-source case, the true source coefficient is defined as $\beta_s = 2\sum_{i=1}^3 e_i$ where $e_i \in \mathbb{R}^p$ is the vector with the $i$-th entry equals to one and all other entries equal to zero. For the fixed-dimensional setting, we estimate the source models using synthetic two-stage regularized regression \citep{tang2023synthetic}, and apply the profiled residual transfer in equation \eqref{eq:sup:profiled_fixed_p}, which is computationally more efficient in low-dimensional problems. \subsection{Sensitivity analysis of the initial estimator}
We show that our method is robust to the choice of the initial estimator $\hat{\beta}_{\text{init}}$, as the condition in Theorem 1 shown. Besides from the used lasso initial estimator in our simulations, we consider some other initial estimators which are the perturbations of the lasso initial estimator. We consider two different perturbation forms, the first has the same sparsity pattern but perturbed non-zero values; the second has some new non-zero elements compared to the lasso initial estimator. In detail, in the first one, we add independent normal distribution perturbations with mean zero and standard deviation 0.5 in each non-zero position. In the second one, we randomly select 50 new position and add this perturbation to those selected position. Results are reported in the Tables \ref{tab:our_only_vary_p_l2},\ref{tab:our_only_vary_p_l1},\ref{tab:our_only_vary_phi_l2},\ref{tab:our_only_vary_phi_l1} as the average of 100 times repeat with other settings the same as the simulation in our main text. The results show that the performance of our method is indeed not sensitive to the selection of initial estimator.

\subsection{Treatment effect of education on earnings (High-dimensional scenario)}
In this section, we continue to analyze the same dataset examining the effect of education on earnings. Unlike the main text, where an initial screening step is used to select several important covariates, here we retain 151 covariates including race, marital status, willingness to work and job satisfaction and apply ProTrans in the high-dimensional setting. All other preprocessing procedures and experimental settings remain the same as in the main text. As in \cite{wang2018bounded}, we treat age, IQ scores, and residence in the South (all measured in 1966) as confounding variables. The high-dimensional linear transfer learning method proposed by \citet{li2022transfer} is applied to the deconfounded data to obtain the oracle estimate $\hat{\eta}_{\text{ora}}$ for $\eta_0 = \beta_t - \beta_s$, which serves as the gold standard. We compare the $l_2$-norm error between the estimated model-shift $\hat{\eta}$ and the gold standard $\hat{\eta}_{\text{ora}}$.

The results, presented in Table \ref{tab:earning_estimation}, indicate that ProTrans is least influenced by confounders compared to the baselines, and has a clear advantage towards treatment effect estimation at the presence of confounders. In all four baselines, covariates such as the amount of liabilities and the father's job level exhibit a significant negative effect on the earnings difference between the treatment and control groups. Conversely, variables related to working hours and the age of individuals living with the respondents show a positive effect. However, both ProTrans and the oracle deconfounded model indicate that these variables do not substantially affect the earnings difference. These variables appear as false positives in the baseline results due to the influence of hidden confounders, whereas ProTrans remains robust against such confounding effects.
